*** Boot

**** Upstream

This attempts to install to an "SD Card", but if it's left on your NVMe, it may
still boot.

+ Depending on references to parts/disks (by UUID or symbol), it maybe possible
  to repack this and push it to the SPI. However, =firewarePartitionOffset=
  implies it's maybe not simple.

See [[https://github.com/0x0013/nixos-rk3588/blob/rkr7.2-pr-integration/examples/upstream-opi/sdcard.nix#L13][examples/upstream-opi/sdcard.nix#L13]] which assumes the interface from
nixpkgs [[https://github.com/NixOS/nixpkgs/blob/658515812992f5867f211dc132b3ad154d26d5ac/nixos/modules/installer/sd-card/sd-image.nix#L37][nixos/modules/installer/sd-card/sd-image.nix#L37]].

#+begin_src nix :tangle nix/machines/theia/boot.nix :mkdirp yes
{ lib, config, pkgs, modulesPath, ... }: let
  rootPartitionUUID = "14e19a7b-0ae0-484d-9d54-43bd6fdc20c7";
  uboot = pkgs.ubootOrangePi5Plus;
in
{

  # pass into devenv.nix instead?
  imports = [ "${modulesPath}/installer/sd-card/sd-image.nix" ];
  boot = {
    kernelParams = [
      "root=UUID=${rootPartitionUUID}"
      "rootfstype=btrfs"
    ];
    consoleLogLevel = 7;

    loader = {
      grub.enable = lib.mkForce false;
      generic-extlinux-compatible.enable = lib.mkForce true;
    };
  };

  #   sd-card.img
  #   │
  #   ├── 32MB gap: contains u-boot code at correct offsets (where board expects it)
  #   ├── 1MB RPi-specific "firmware" partition (unused; mounts to /boot/firmware)
  #   └── / 
  #       ├── /boot/extlinux: how u-boot finds the kernel to boot
  #       └── /boot/nixos: all installed kernels, referenced by extlinux

  sdImage = {
    inherit rootPartitionUUID;
    compressImage = true;

    populateFirmwareCommands = "";
    firmwareSize = 1; # MiB
    firmwarePartitionName = "DUMMY_UNUSED";
    firmwarePartitionOffset = 32; # needed to fit idbloader.img & u-boot.itb

    populateRootCommands = ''
      ${config.boot.loader.generic-extlinux-compatible.populateCmd} -c ${config.system.build.toplevel} -d ./files/boot
    '';

    postBuildCommands = ''
      dd if=${uboot}/idbloader.img of=$img seek=64 conv=fsync,notrunc
      dd if=${uboot}/u-boot.itb of=$img seek=16384 conv=fsync,notrunc
    '';
  };

  fileSystems."/".fsType = lib.mkForce "btrfs";
  fileSystems."/".device = lib.mkForce rootPartitionUUID;
}
